# 如何新增工具

本指南演示如何在 AgentTemplate 中创建新的工具。

## 内置工具 vs Agent 专用工具

- **内置工具**：通用工具，放在 `src/app/tools/builtin/`
- **Agent 专用工具**：特定 Agent 使用，放在该 Agent 目录下

## 创建简单工具

### 1. 定义工具函数

```python
# src/app/tools/builtin/time.py
"""时间工具：获取当前时间、格式化、解析。"""

from datetime import datetime, timezone
from zoneinfo import ZoneInfo
from pydantic import BaseModel, Field

from app.tools.governance import governed_tool


class GetCurrentTime(BaseModel):
    """获取当前时间。"""
    timezone: str = Field(
        default="UTC",
        description="时区，如 'UTC'、'Asia/Shanghai'、'America/New_York'"
    )


@governed_tool(scopes=["tools:time"])
async def current_time(params: GetCurrentTime) -> dict:
    """获取指定时区的当前时间。
    
    这是一个只读工具，不需要人工审批。
    """
    try:
        tz = ZoneInfo(params.timezone)
    except Exception:
        return {
            "error": "invalid_timezone",
            "message": f"时区 '{params.timezone}' 不存在",
        }
    
    now = datetime.now(tz)
    
    return {
        "timezone": params.timezone,
        "iso": now.isoformat(),
        "timestamp": now.timestamp(),
        "year": now.year,
        "month": now.month,
        "day": now.day,
        "hour": now.hour,
        "minute": now.minute,
        "second": now.second,
        "weekday": now.strftime("%A"),
    }
```

### 2. 注册到工具注册表

```python
# src/app/tools/registry.py
# 在 __init__ 方法中注册

from app.tools.builtin.time import current_time

self.register("current_time", current_time)
```

### 3. 使用工具

```python
# 在 Agent 中获取
tools = ctx.tools.langchain(["current_time"])
```

## 创建写工具（需要审批）

写工具默认需要人工审批：

```python
from app.tools.governance import governed_tool
from app.runtime.contracts import current_run


class DeleteFile(BaseModel):
    """删除文件参数。"""
    path: str = Field(description="要删除的文件路径")
    confirm: bool = Field(
        default=False,
        description="确认删除（必须设为 true）"
    )


@governed_tool(
    scopes=["tools:files:write"],
    requires_approval=True,  # 显式标记需要审批（写工具默认如此）
)
async def delete_file(params: DeleteFile) -> dict:
    """删除指定文件。
    
    这是一个写操作，需要人工审批。副作用通过 tool_call_id 自动幂等。
    """
    if not params.confirm:
        return {
            "error": "confirmation_required",
            "message": "必须设置 confirm=true 才能删除",
        }
    
    # 从运行上下文获取配置
    context = current_run.get()
    if not context:
        return {"error": "no_context", "message": "工具只能在运行上下文中使用"}
    
    # 实现删除逻辑...
    import os
    from pathlib import Path
    
    try:
        # 安全检查（路径越界、符号链接等）
        # ... 实现细节 ...
        
        os.remove(path)
        
        return {
            "success": True,
            "path": params.path,
            "message": f"文件 {params.path} 已删除",
        }
    except FileNotFoundError:
        return {
            "error": "file_not_found",
            "message": f"文件 {params.path} 不存在",
        }
    except Exception as error:
        return {
            "error": "delete_failed",
            "message": str(error),
        }
```

### 人工审批流程

1. Agent 调用写工具
2. 执行中断（`interrupt`）
3. 返回给客户端，等待审批
4. 客户端提交审批决定：

```python
# 批准
POST /api/v1/threads/{thread_id}/runs/wait
{
  "agent_id": "assistant",
  "resume": {"action": "approve"},
  "expected_sequence": 5
}

# 拒绝
POST /api/v1/threads/{thread_id}/runs/wait
{
  "agent_id": "assistant",
  "resume": {"action": "reject", "reason": "不安全的操作"},
  "expected_sequence": 5
}
```

## 访问运行上下文

工具可以通过 `current_run.get()` 访问运行上下文：

```python
from app.runtime.contracts import current_run


@governed_tool(scopes=["tools:custom"])
async def custom_tool(params: CustomParams) -> dict:
    context = current_run.get()
    if not context:
        return {"error": "no_context"}
    
    # 访问身份
    tenant_id = context.principal.tenant_id
    subject_id = context.principal.subject_id
    
    # 访问预算
    remaining_budget = context.budget.cost - context.usage.get("total_cost", 0)
    
    # 访问运行信息
    run_id = context.run_id
    thread_id = context.thread_id
    agent_id = context.agent_id
    
    # ... 使用上下文信息 ...
    
    return {"result": "..."}
```

## 结构化错误返回

工具应返回结构化结果，而不是抛出异常：

```python
@governed_tool(scopes=["tools:api"])
async def call_external_api(params: ApiParams) -> dict:
    """调用外部 API。"""
    try:
        # API 调用逻辑...
        return {
            "success": True,
            "data": response_data,
        }
    except TimeoutError:
        return {
            "error": "timeout",
            "message": "API 请求超时",
            "retry_after": 60,
        }
    except ValueError as e:
        return {
            "error": "validation_error",
            "message": str(e),
        }
    except Exception as e:
        return {
            "error": "unknown_error",
            "message": str(e),
        }
```

## 使用配置

工具可以通过依赖注入获取配置：

```python
# src/app/tools/builtin/custom.py
from app.core.config import ToolConfig


class CustomToolRegistry:
    """自定义工具注册表，持有配置。"""
    
    def __init__(self, config: ToolConfig):
        self.config = config
    
    def create_tool(self):
        """创建工具闭包，捕获配置。"""
        config = self.config
        
        @governed_tool(scopes=["tools:custom"])
        async def custom_tool(params: CustomParams) -> dict:
            # 使用 config
            timeout = config.timeout_seconds
            # ...
            return {"result": "..."}
        
        return custom_tool


# 在 ToolRegistry.__init__ 中
from app.tools.builtin.custom import CustomToolRegistry

custom_registry = CustomToolRegistry(self.config)
self.register("custom_tool", custom_registry.create_tool())
```

## Agent 专用工具

对于只有特定 Agent 使用的工具，放在该 Agent 目录下：

```python
# src/app/agents/assistant/tools.py
"""助手专用工具。"""

from pydantic import BaseModel
from app.tools.governance import governed_tool


class CalculateParams(BaseModel):
    expression: str


@governed_tool(scopes=["tools:calculate"])
async def calculate(params: CalculateParams) -> dict:
    """安全的数学计算工具。"""
    try:
        # 使用 ast.literal_eval 或其他安全方式
        result = eval(params.expression, {"__builtins__": {}}, {})
        return {
            "expression": params.expression,
            "result": result,
        }
    except Exception as e:
        return {
            "error": "calculation_failed",
            "message": str(e),
        }


# 在 agent.py 中使用
async def build(ctx: BuildContext):
    # 导入专用工具
    from .tools import calculate
    
    # 手动转换为 LangChain 工具
    from langchain_core.tools import StructuredTool
    
    calc_tool = StructuredTool.from_function(
        func=calculate,
        name="calculate",
        description="执行数学计算",
    )
    
    # 获取通用工具
    builtin_tools = ctx.tools.langchain(["read_file"])
    
    all_tools = builtin_tools + [calc_tool]
    
    return create_agent(
        model=await ctx.models.chat("gpt-4o"),
        tools=all_tools,
        system_message="...",
        checkpointer=ctx.checkpointer,
    )
```

## 工具治理配置

### Scope 要求

工具的 `scopes` 参数定义了调用该工具所需的权限：

```python
@governed_tool(scopes=["tools:files:read"])
async def read_file(params):
    ...

@governed_tool(scopes=["tools:files:write"])
async def write_file(params):
    ...
```

API Key 必须包含对应的 scope：

```yaml
auth:
  api_keys:
    - digest: "..."
      scopes:
        - tools:files:read   # 可以读
        - tools:files:write  # 可以写
```

### 审批策略

默认行为：
- **读工具**：不需要审批
- **写工具**：需要审批

显式控制：

```python
@governed_tool(
    scopes=["tools:custom"],
    requires_approval=False,  # 强制不需要审批（即使是写操作）
)
async def safe_write(params):
    ...
```

### 超时控制

工具超时由 `tools.timeout_seconds` 控制（默认 30 秒）。

### 调用次数限制

通过中间件限制工具调用次数：

```python
# 在 Agent 构建时
from langgraph.prebuilt import create_react_agent

agent = create_react_agent(
    model=model,
    tools=tools,
    checkpointer=ctx.checkpointer,
)

# LangGraph 自带 recursion_limit
# 或使用自定义中间件
```

## 安全考虑

### 路径安全

```python
import os
from pathlib import Path


def safe_open(base_dir: Path, relative_path: str):
    """安全地打开路径，防止越界。"""
    full_path = (base_dir / relative_path).resolve()
    
    # 检查是否在允许目录内
    if not full_path.is_relative_to(base_dir):
        raise ValueError("路径越界")
    
    # 使用 O_NOFOLLOW 拒绝符号链接
    fd = os.open(full_path, os.O_RDONLY | os.O_NOFOLLOW)
    return os.fdopen(fd, 'r')
```

### 网络安全

```python
import ipaddress
from urllib.parse import urlparse


def is_safe_url(url: str, allow_private: bool = False) -> bool:
    """检查 URL 是否安全。"""
    parsed = urlparse(url)
    
    # 只允许 HTTP/HTTPS
    if parsed.scheme not in {"http", "https"}:
        return False
    
    # 不允许内嵌凭据
    if parsed.username or parsed.password:
        return False
    
    # 解析 IP
    try:
        ip = ipaddress.ip_address(parsed.hostname)
        if not allow_private and not ip.is_global:
            return False
    except ValueError:
        pass  # 域名，需要进一步解析
    
    return True
```

### SQL 安全

```python
# 使用参数化查询
async def safe_query(conn, table: str, conditions: dict):
    """安全的查询构造。"""
    # 验证表名在允许列表中
    allowed_tables = ["users", "orders"]
    if table not in allowed_tables:
        raise ValueError(f"表 {table} 不在允许列表")
    
    # 使用参数化
    placeholders = " AND ".join(f"{k} = ${i+1}" for i, k in enumerate(conditions))
    query = f"SELECT * FROM {table} WHERE {placeholders}"
    
    return await conn.fetch(query, *conditions.values())
```

## 常见问题

### 工具未注册

检查 `ToolRegistry.__init__` 中是否调用了 `self.register()`。

### 权限不足

检查 API Key 的 scopes 是否包含工具所需的权限。

### 审批流程卡住

确认客户端正确处理 `interrupt` 事件并提交 `resume`。

### 幂等性问题

LangGraph 自动基于 `tool_call_id` 实现幂等，无需工具显式处理。
